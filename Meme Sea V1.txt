--=========================== HELPERS ===========================
local function Alive(o)
    local h = o and o:FindFirstChild("Humanoid")
    return h and h.Health > 0
end

local function GoTo(cf)
    local c = LP.Character
    if Alive(c) then c:SetPrimaryPartCFrame(cf) end
end

local function MoveTo(cf)
    local c = LP.Character
    if Alive(c) then c:MoveTo(cf.p) end
end

local function Data(name)
    local v = PlayerData:FindFirstChild(name)
    return v and v.Value or 0
end

local function Material(name)
    local s = Items.ItemStorage
    local v = s:FindFirstChild(name)
    return v and v.Value or 0
end

local function HasSword(name)
    local w = Items.Weapon
    local v = w:FindFirstChild(name)
    return v and v.Value > 0
end

local function HasAbility(name)
    local a = AbilityF:FindFirstChild(name)
    return a and a.Value
end

local function EquipWeapon()
    local bp, ch = LP:FindFirstChild("Backpack"), LP.Character
    if Alive(ch) and bp then
        for _, t in ipairs(bp:GetChildren()) do
            if t:IsA("Tool") and t.ToolTip:find(Config.Tool) then
                ch.Humanoid:EquipTool(t)
            end
        end
    end
end

local function Attack()
    local ch = LP.Character
    if not ch then return end
    if Config.AutoAttack then
        VU:CaptureController()
        VU:Button1Down(Vector2.new(1e4, 1e4))
    end
    if Config.AutoAura and HasAbility("Aura") then
        local folder = ch:FindFirstChild("AuraColor_Folder")
        if folder and #folder:GetChildren() < 1 then
            pcall(function() MainEvents.Ability:InvokeServer("Aura") end)
        end
    end
end

local function NextEnemy(name)
    for _, m in ipairs(Monsters:GetChildren()) do
        if (not name or m.Name == name) and Alive(m) then return m end
    end
    return nil
end

local function BringMobs(name, cf, all)
    for _, m in ipairs(Monsters:GetChildren()) do
        if (all or m.Name == name) and Alive(m) and m.PrimaryPart then
            if (m.PrimaryPart.Position - cf.p).Magnitude < 500 then
                local h, pp = m:FindFirstChild("Humanoid"), m.PrimaryPart
                h.WalkSpeed = 0
                h:ChangeState(14)
                pp.CFrame = cf
                pp.CanCollide = false
                pp.Size = Vector3.new(50, 50, 50)
            end
        end
    end
    pcall(function() sethiddenproperty(LP, "SimulationRadius", math.huge) end)
end

local function Kill(enemy, all)
    local e = typeof(enemy) == "Instance" and enemy or NextEnemy(enemy)
    if Alive(e) and e.PrimaryPart then
        GoTo(e.PrimaryPart.CFrame * FarmCF())
        EquipWeapon()
        if not e:FindFirstChild("Reverse_Mark") then Attack() end
        if Config.BringMobs then BringMobs(typeof(enemy) == "string" and enemy or nil, e.PrimaryPart.CFrame, all) end
        return true
    end
end

local function TakeQuest(npc, cf, delay)
    local q = QuestsNPC:FindFirstChild(npc)
    if q and LP:DistanceFromCharacter(q.WorldPivot.p) < 6 then
        local prompt = q:FindFirstChild("Block") and q.Block:FindFirstChild("QuestPrompt")
        if prompt then pcall(fireproximityprompt, prompt) end
        task.wait(delay or 0.15)
        return true
    end
    MoveTo(cf or QuestLocation[npc].CFrame)
end

local function ClearQuests(keep)
    for _, q in ipairs(QuestFolder:GetChildren()) do
        if q.QuestGiver.Value ~= keep and q.Target.Value ~= "None" then
            pcall(function()
                QuestEvents.Quest:FireServer("Abandon_Quest", { QuestSlot = q.Name })
            end)
        end
    end
end

local function ActiveQuest()
    for _, q in ipairs(QuestFolder:GetChildren()) do
        if q.Target.Value ~= "None" then return q end
    end
end

local function OnQuest(npc)
    local q = ActiveQuest()
    return q and q.QuestGiver.Value == npc
end

local function Summon(pos, promptPath)
    if LP:DistanceFromCharacter(pos) < 6 then
        local o = Island
        for _, p in ipairs(promptPath) do
            o = o and o:FindFirstChild(p)
        end
        if o then pcall(fireproximityprompt, o) end
        return true
    end
    MoveTo(CFrame.new(pos))
    return true
end

--=========================== ESP ===========================
local espConns = {}
local function ClearESP()
    for _, c in ipairs(espConns) do pcall(function() c:Disconnect() end) end
    espConns = {}
    for _, m in ipairs(Monsters:GetChildren()) do
        for _, ch in ipairs(m:GetChildren()) do
            if ch.Name == "SigmaESP" or ch.Name == "SigmaHL" then ch:Destroy() end
        end
    end
end

local function TagESP(m)
    if not m:IsA("Model") or not m.PrimaryPart then return end
    local hl = Instance.new("Highlight")
    hl.Name = "SigmaHL"
    hl.FillColor = Color3.fromRGB(255, 60, 60)
    hl.FillTransparency = 0.65
    hl.OutlineColor = Color3.fromRGB(255, 255, 255)
    hl.Parent = m

    local bb = Instance.new("BillboardGui")
    bb.Name = "SigmaESP"
    bb.Size = UDim2.fromOffset(200, 40)
    bb.AlwaysOnTop = true
    bb.StudsOffsetWorldSpace = Vector3.new(0, 4, 0)
    bb.Parent = m

    local txt = Instance.new("TextLabel")
    txt.Size = UDim2.fromScale(1, 1)
    txt.BackgroundTransparency = 1
    txt.Font = Enum.Font.GothamBold
    txt.TextSize = 13
    txt.TextColor3 = Color3.fromRGB(255, 235, 120)
    txt.TextStrokeTransparency = 0
    txt.Text = m.Name
    txt.Parent = bb

    table.insert(espConns, RunService.RenderStepped:Connect(function()
        if not m.Parent then return end
        local ch = LP.Character
        local root = ch and ch:FindFirstChild("HumanoidRootPart")
        if root and m.PrimaryPart then
            txt.Text = ("%s [%d]"):format(m.Name, math.floor((m.PrimaryPart.Position - root.Position).Magnitude))
        end
    end))
end

local function EnableESP()
    for _, m in ipairs(Monsters:GetChildren()) do TagESP(m) end
    table.insert(espConns, Monsters.ChildAdded:Connect(function(m)
        if Config.ESP then task.wait(0.2); TagESP(m) end
    end))
end

        local b = Monsters:FindFirstChild("Meme Beast") or rsMonsterSpawn:FindFirstChild("Meme Beast")
        if b then GoTo(b.WorldPivot); EquipWeapon(); Attack(); return true end
    end },
    { flag = "LordSus", run = function()

    end },
    { flag = "EvilNoob", run = function()

        if b then
            if not OnQuest("Floppa Quest 29") and Level.Value >= 1400 then
                ClearQuests("Floppa Quest 29"); TakeQuest("Floppa Quest 29", nil, 1)
            else Kill(b) end
            return true
        elseif Material("Noob Head") > 0 then
            return Summon(Vector3.new(-2356, -81, 3180), {"MoaiIsland", "Summon2", "Summon", "SummonPrompt"})
        end
    end },
    { flag = "Pumpkin", run = function()
        local b = Monsters:FindFirstChild("Giant Pumpkin") or rsMonsterSpawn:FindFirstChild("Giant Pumpkin")
        if b then
            if not OnQuest("Floppa Quest 23") and Level.Value >= 1100 then
                ClearQuests("Floppa Quest 23"); TakeQuest("Floppa Quest 23", nil, 1)
            else Kill(b) end
            return true
        elseif Material("Flame Orb") > 0 then
            return Summon(Vector3.new(-1180, -93, 1462), {"PumpkinIsland", "Summon1", "Summon", "SummonPrompt"})
        end
    end },
    { flag = "Floppa", run = function()
        if HasSword("Floppa") then return end
        if OnQuest("Cool Floppa Quest") then
            local lava = Island:FindFirstChild("FloppaIsland") and Island.FloppaIsland:FindFirstChild("Lava Floppa")
            if not lava then return end
            if LP:DistanceFromCharacter(lava.WorldPivot.p) < 12 then
                pcall(fireproximityprompt, lava.ClickPart.ProximityPrompt)
            else MoveTo(CFrame.new(794, -31, -440)) end
            return true
        end
        ClearQuests("Cool Floppa Quest")
        TakeQuest("Cool Floppa Quest", CFrame.new(758, -31, -424))
        return true
    end },
    { flag = "RaceOrb", run = function()
        if Level.Value < 500 then return end
        if OnQuest("Dancing Banana Quest") then
            if Kill("Sogga") then return true end
            MoveTo(EnemyLocation.Sogga.CFrame)
            return true
        end
        ClearQuests("Dancing Banana Quest")
        TakeQuest("Dancing Banana Quest", CFrame.new(-2620, -80, -2001))
        return true
    end },
    { flag = "Raid", run = function()
        if Level.Value < 1000 then return end
        local map
        for _, r in ipairs(workspace:FindFirstChild("Raids"):GetChildren()) do
            if r.Joiners:FindFirstChild(LP.Name) then map = r break end
        end
        if map then
            if map:GetAttribute("Starting") ~= 0 then
                MiscEvents.StartRaid:FireServer("Start"); task.wait(1)
            else
                local e
                for _, m in ipairs(Monsters:GetChildren()) do
                    if m:GetAttribute("Raid_Enemy") and Alive(m) then e = m break end
                end
                if e then Kill(e, true) else
                    local s = map:FindFirstChild("Spawn_Location")
                    if s then MoveTo(s.CFrame) end
                end
            end
        else
            local area = Region:FindFirstChild("RaidArea")
            if area then MoveTo(CFrame.new(area.Position)) end
        end
        return true
    end },
    { flag = "Selected", run = function()
        local e = Config.SelectedEnemy
        if not e then return end
        local npc
        for _, q in ipairs(QuestFolder:GetChildren()) do end
        if Kill(e) then return true end
        local loc = EnemyLocation:FindFirstChild(e)
        if loc then MoveTo(loc.CFrame) end
        return true
    end },
    { flag = "Level", run = function()
        local current = ActiveQuest()
        if current then
            local npc = current.QuestGiver.Value
            if Kill(current.Target.Value) then return true end
            local loc = QuestLocation:FindFirstChild(npc)
            if loc then MoveTo(loc.CFrame) end
            return true
        end
        for _, q in ipairs(QuestFolder:GetChildren()) do end
        -- pega a próxima quest pelo nível
        local best
        for _, npc in ipairs(QuestsNPC:GetChildren()) do
            local loc = QuestLocation:FindFirstChild(npc.Name)
            if loc and Level.Value < MaxLevel then best = npc.Name end
        end
        if best then TakeQuest(best) return true end
    end },
}

local function FarmLoop()
    while task.wait() do
        for _, job in ipairs(Jobs) do
            if Flags[job.flag] then
                local ok, res = pcall(job.run)
                if ok and res then break end
            end
        end
    end
end

--=========================== AUTO STATS ===========================
local StatMap = {
    Power  = "MemePowerLevel",
    Health = "DefenseLevel",
    Weapon = "SwordLevel",
    Melee  = "MeleeLevel",
}

local function StatsLoop()
    while task.wait(0.6) do
        if Config.AutoStats then
            for label, key in pairs(StatMap) do
                if Config.Stats[label] and SkillPoint.Value > 0 then
                    local v = PlayerData[key]
                    if v and v.Value < MaxLevel then
                        pcall(function()
                            MainEvents.StatsFunction:InvokeServer({
                                Target = key,
                                Action = "UpgradeStats",
                                Amount = math.clamp(Config.AutoStatsPts, 0, MaxLevel - v.Value),
                            })
                        end)
                    end
                end
            end
        end
    end
end

local function N(class, props)
    local i = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then i[k] = v end
    end
    if props and props.Parent then i.Parent = props.Parent end
    return i
end

        Size = UDim2.fromOffset(14, 14), Position = UDim2.fromOffset(state and 22 or 2, 2),
        BackgroundColor3 = Color3.new(1, 1, 1), Parent = pill,

    end)

end

            BackgroundColor3 = T.panel, Text = tostring(opt), Font = Enum.Font.Gotham,
            TextSize = 11, TextColor3 = T.sub, AutoButtonColor = false, Parent = list,

local function Slider(parent, text, min, max, default, callback)
    local row = N("Frame", { Size = UDim2.new(1, 0, 0, 40), BackgroundColor3 = T.card,
        BorderSizePixel = 0, Parent = parent })
    corner(row, 8)
    local label = N("TextLabel", {
        Size = UDim2.new(1, -20, 0, 20), Position = UDim2.fromOffset(10, 2),
        BackgroundTransparency = 1, Text = text .. ": " .. default, Font = Enum.Font.Gotham,
        TextSize = 11, TextColor3 = T.sub, TextXAlignment = Enum.TextXAlignment.Left, Parent = row,
    })
    local track = N("Frame", {
        Size = UDim2.new(1, -20, 0, 6), Position = UDim2.new(0, 10, 1, -12),
        BackgroundColor3 = T.line, Parent = row,
    })
    corner(track, 3)
    local fill = N("Frame", { Size = UDim2.fromScale((default - min) / (max - min), 1),
        BackgroundColor3 = T.acc, Parent = track })
    corner(fill, 3)
    local dragging = false
    local function set(x)
        local rel = math.clamp((x - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        local val = math.floor(min + (max - min) * rel + 0.5)
        fill.Size = UDim2.fromScale(rel, 1)
        label.Text = text .. ": " .. val
        callback(val)
    end
    track.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true; set(i.Position.X)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            set(i.Position.X)
        end
    end)
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = false end
    end)
end

local function fade(obj, toVisible)
    obj.Visible = true
    local tw = TweenService:Create(obj, TweenInfo.new(0.15), { BackgroundTransparency = toVisible and 0 or 1 })
    tw:Play()
    if not toVisible then tw.Completed:Wait(); obj.Visible = false end
end
